\section*{Modular arithmetic}
\begin{lstlisting}
const ll MOD = 1'000'000'007LL;
ll power(ll b, ll e, ll m = MOD) {      // b^e mod m, O(log e)
    ll r = 1 % m; b %= m; if (b < 0) b += m;
    for (; e > 0; e >>= 1, b = b * b % m)
        if (e & 1) r = r * b % m;
    return r;
}
ll inv(ll a) { return power(a, MOD - 2); }   // MOD prime, a % MOD != 0
\end{lstlisting}

\section*{Sieve (smallest prime factor)}
\begin{lstlisting}
vector<int> spf(N + 1, 0);
for (int i = 2; i <= N; i++)
    if (!spf[i])
        for (int j = i; j <= N; j += i)
            if (!spf[j]) spf[j] = i;
// factorise x: while (x > 1) { int p = spf[x]; while (x % p == 0) x /= p; }
\end{lstlisting}

\section*{Xor basis}
\begin{lstlisting}
unsigned long long b[61] = {};
bool insert(unsigned long long v) {      // true if v was independent
    for (int i = 60; i >= 0; i--) if (v >> i & 1) {
        if (!b[i]) { b[i] = v; return true; }
        v ^= b[i];
    }
    return false;                        // v is representable
}
\end{lstlisting}

\section*{Greedy proof patterns}
\begin{itemize}
  \item \textbf{Exchange:} swap the first disagreement of an optimal solution
        into the greedy choice; show nothing gets worse.
  \item \textbf{Stays ahead:} after every step greedy is at least as good by
        some measure.
  \item \textbf{Lower bound = construction:} find a quantity every solution
        must pay; show greedy pays exactly that.
\end{itemize}
